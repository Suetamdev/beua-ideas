# Event 2
country_event = {
	id = beua_elven_supremacy_ideas.2
	title = beua_elven_supremacy_ideas.2.t
	desc = beua_elven_supremacy_ideas.2.d
	
	picture = COURT_eventPicture

	is_triggered_only = yes

	trigger = {
		has_idea_group = beua_elven_supremacy_ideas
		has_country_modifier = elven_administration
        any_owned_province = {
		    has_human_minority_trigger = yes
            NOT = { has_province_modifier = beua_elven_supremacy_ideas.2.1.m }
		}
	}

    immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
                    has_human_minority_trigger = yes
                    NOT = { has_province_modifier = beua_elven_supremacy_ideas.2.1.m }
				}
				set_province_flag = beua_elven_supremacy_ideas.2.f
			}
		}
	}

	mean_time_to_happen = {
		days = 1
	}
	
	option = {
		name = beua_elven_supremacy_ideas.2.1
		add_estate_nobles_loyalty_effect = yes
        every_owned_province = {
			limit = {
				has_province_flag = beua_elven_supremacy_ideas.2.f
			}
			add_unrest = 5
			add_province_modifier = {
				name = beua_elven_supremacy_ideas.2.1.m
				duration = 3650
			}
		}
	}

    option = {
		name = beua_elven_supremacy_ideas.2.2
        reduce_estate_nobles_loyalty_effect = yes
		add_prestige = -15
	}

    after = {
		hidden_effect = {
			every_owned_province = {
				limit = {
					has_province_flag = beua_elven_supremacy_ideas.2.f
				}
				clr_province_flag = beua_elven_supremacy_ideas.2.f
			}
		}
	}
}
