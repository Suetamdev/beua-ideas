# Event 2
country_event = {
	id = beua_industrial_ideas_event.2
	title = beua_industrial_ideas_event.2.t
	desc = beua_industrial_ideas_event.2.d

	picture = ANGRY_MOB_eventPicture
	
	is_triggered_only = yes

	trigger = {
		has_idea_group = beua_industrial_ideas
		any_owned_province = {
			has_manufactory_trigger = yes
			is_state = yes
			NOT = { has_province_modifier = beua_industrial_ideas_event.2.1.m }
		}
	}

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					has_manufactory_trigger = yes
					is_state = yes
					NOT = { has_province_modifier = beua_industrial_ideas_event.2.1.m }
				}
				set_province_flag = beua_industrial_ideas_event.2.f
			}
		}
	}

	mean_time_to_happen = { 
		days = 1 
	}

	option = {
		ai_chance = { factor = 50 }
		name = beua_industrial_ideas_event.2.1
		every_owned_province = {
			limit = {
				has_province_flag = beua_industrial_ideas_event.2.f
			}
			add_unrest = 5
			add_province_modifier = {
				name = beua_industrial_ideas_event.2.1.m
				duration = 3650
			}
		}
	}

	option = {
		ai_chance = { factor = 50 }
		name = beua_industrial_ideas_event.2.2
		add_treasury = -300
	}

	option = {
		ai_chance = { factor = 100 }
		name = beua_industrial_ideas_event.2.3
		highlight = yes
		trigger = {
			has_faction = nl_cooperatists
		}
		add_manpower = -5.
	}

	after = {
		hidden_effect = {
			every_owned_province = {
				limit = {
					has_province_flag = beua_industrial_ideas_event.2.f
				}
				clr_province_flag = beua_industrial_ideas_event.2.f
			}
		}
	}
}

country_event = {
	id = beua_industrial_ideas_event.3
	title = beua_industrial_ideas_event.3.t
	desc = beua_industrial_ideas_event.3.d

	picture = COUNTRY_COLLAPSE_eventPicture

	is_triggered_only = yes

	trigger = {
		has_idea_group = beua_industrial_ideas
		any_owned_province = {
			has_manufactory_trigger = yes
			is_state = yes
			OR = {
				trade_goods = cloth
				trade_goods = silk
				trade_goods = wool
				trade_goods = dyes
			}
		}
		NOT = {
			OR = {
				has_country_modifier = beua_industrial_ideas_event.3.1.m
				has_country_modifier = beua_industrial_ideas_event.3.2.m
				has_country_modifier = beua_industrial_ideas_event.3.3.m
				has_country_modifier = beua_industrial_ideas_event.3.4.m
			}
		}
	}

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					has_manufactory_trigger = yes
					is_state = yes
					OR = {
						trade_goods = cloth
						trade_goods = silk
						trade_goods = wool

				set_province_flag = beua_industrial_ideas_event.3.f

		days = 1 
	}

		name = beua_industrial_ideas_event.3.1

		ai_chance = { factor = 50 }
		add_country_modifier = {
			name = "beua_industrial_ideas_event.3.2.m"
			duration = 3650
		}
		every_owned_province = {
			limit = {
				has_province_flag = beua_industrial_ideas_event.3.f
			}
			add_devastation = 20
			clr_province_flag = beua_industrial_ideas_event.3.f
		}
	}

	option = {
		name = beua_industrial_ideas_event.3.3
		ai_chance = { factor = 100 }
		highlight = yes
		trigger = {
			has_faction = nl_cooperatists
		}
		add_country_modifier = {
			name = "beua_industrial_ideas_event.3.3.m"

		}

			}
			add_devastation = 20
