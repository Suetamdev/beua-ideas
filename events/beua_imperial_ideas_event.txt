# Event 1
country_event = {
	id = beua_imperial_ideas_event.1
	title = beua_imperial_ideas_event.1.t
	desc = beua_imperial_ideas_event.1.d

	picture = PROTESTANT_GREAT_REFORMER_eventPicture
	
	is_triggered_only = yes

	trigger = {
		has_idea_group = beua_imperial_ideas
		NOT = {
			OR = { 
				tag = A11
				tag = A45
                has_country_modifier = human_administration
			}
		}
        OR = {
            A11 = {
                is_elector = yes
            }
            A45 = {
                is_elector = yes
            }
        }
	}

	mean_time_to_happen = { 
        days = 1 
    }

	option = {
		name = beua_imperial_ideas_event.1.1
        if = {
			limit = {
				A11 = {
					is_elector = yes
				}
			}
			reverse_add_opinion = {
				who = A11
				modifier = beua_imperial_ideas_event.1.1.m
			}
		}
        if = {
			limit = {
				A45 = {
					is_elector = yes
				}
			}
			reverse_add_opinion = {
				who = A45
				modifier = beua_imperial_ideas_event.1.1.m
			}
		}
		medium_increase_of_human_tolerance_effect = yes
		add_adm_power = -25
		add_dip_power = -25
		add_mil_power = -25
	}
	option = {
		name = beua_imperial_ideas_event.1.2
		add_prestige = 10
	}
}

country_event = {
	id = beua_imperial_ideas_event.7
	title = beua_imperial_ideas_event.7.t
	desc = beua_imperial_ideas_event.7.d

	picture = DEBATE_REPUBLICAN_eventPicture
	
	is_triggered_only = yes

	trigger = {
		has_idea_group = beua_imperial_ideas
		NOT = {	is_emperor = yes }
	}

	immediate = {
		hidden_effect = {
			random_country = {
				limit = { is_emperor = yes }
				save_event_target_as = beua_imperial_ideas_event_7_c
			}
		}
	}

	mean_time_to_happen = { 
        days = 1 
    }

	option = {
		name = beua_imperial_ideas_event.7.1
		event_target:beua_imperial_ideas_event_7_c = {
			add_imperial_influence = 1
			add_opinion = {
				who = ROOT
				modifier = beua_imperial_ideas_event.7.1.m
			}
		}
		add_dip_power = 50
	}

	option = {
		name = beua_imperial_ideas_event.7.2
		event_target:beua_imperial_ideas_event_7_c = {
			add_opinion = {
				who = ROOT
				modifier = beua_imperial_ideas_event.7.2.m
			}
		}
		add_dip_power = 100
	}
}

	is_triggered_only = yes

        NOT = {
            OR = {
                tag = A25
                has_country_modifier = beua_imperial_ideas_event.9.1.m.2
            } 
        }
        A25 = {
            is_elector = yes
        }
        any_owned_province = {
			province_has_center_of_trade_of_level = 1
			is_state = yes
            NOT = { has_province_modifier = beua_imperial_ideas_event.9.1.m.2 }
		}
	}
	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = { 
					province_has_center_of_trade_of_level = 1 
					is_state = yes
                    NOT = { has_province_modifier = beua_imperial_ideas_event.9.1.m.2 }
				}
				save_event_target_as = beua_imperial_ideas_event_9_c
			}
		}
	}

	mean_time_to_happen = { 
        days = 1 
    }

	option = {
		name = beua_imperial_ideas_event.9.1
		reverse_add_opinion = {
			who = A25
			modifier = beua_imperial_ideas_event.9.1.m.1
		}
        event_target:beua_imperial_ideas_event_9_c = {
			add_province_modifier = {
				name = "beua_imperial_ideas_event.9.1.m.2"
				duration = 3650
			}
		}
	}
    
	option = {
		name = beua_imperial_ideas_event.9.2
