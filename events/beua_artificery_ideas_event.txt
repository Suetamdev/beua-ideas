# Event 10
country_event = {
	id = beua_artificery_ideas_event.10
	title = beua_artificery_ideas_event.10.t
	desc = beua_artificery_ideas_event.10.d

	picture = SPELL_LIST_eventPicture
	
	is_triggered_only = yes

	trigger = {
		has_idea_group = beua_artificery_ideas
		has_estate = estate_artificers
        NOT = { has_country_flag = beua_artificery_ideas_event.10.f }
		any_owned_province = {
            OR = {
			    has_province_modifier = old_hold
                has_province_modifier = artificier_hall
                development = 90
            }
            OR = {
				has_terrain = dwarven_hold
				has_terrain = dwarven_hold_surface
			}
            NOT = { has_province_modifier = beua_artificery_ideas_event.10.1.m }
		}
	}

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					OR = {
                        has_province_modifier = old_hold
                        has_province_modifier = artificier_hall
                        development = 90
                    }
                    OR = {
                        has_terrain = dwarven_hold
                        has_terrain = dwarven_hold_surface
                    }
                    NOT = { has_province_modifier = beua_artificery_ideas_event.10.1.m }
				}
				save_event_target_as = beua_artificery_ideas_event_10_c
			}
		}
	}

	mean_time_to_happen = { 
		days = 1
        modifier = {
			factor = 10
			any_owned_province = {
                OR = {
                    has_province_modifier = old_hold
                    has_province_modifier = artificier_hall
                    development = 90
                }
                OR = {
                    has_terrain = dwarven_hold
                    has_terrain = dwarven_hold_surface
                }
		    }
			NOT = { has_country_flag = beua_artificery_ideas_event.10.f }
		}
	}

	option = {
		name = beua_artificery_ideas_event.10.1
		event_target:beua_artificery_ideas_event_10_c = {
			change_trade_goods = precursor_relics
            add_base_tax = -3
            add_base_production = -3
            add_base_manpower = -3
            add_province_modifier = {
                name = beua_artificery_ideas_event.10.1.m
                duration = -1
            }
		}
	}

	option = {
		name = beua_artificery_ideas_event.10.2
	}

    after = {
		set_country_flag = beua_artificery_ideas_event.10.f
	}
}
