# Event 4
country_event = {
	id = beua_monstrous_ideas_event.4
	title = beua_monstrous_ideas_event.4.t
	desc = beua_monstrous_ideas_event.4.d

	picture = CITY_DEVELOPMENT_eventPicture
	
	is_triggered_only = yes

	trigger = {
		has_idea_group = beua_monstrous_ideas
		NOT = { low_tolerance_goblin_race_trigger = yes }
		any_owned_province = {
			NOT = {
				OR = {
					has_province_modifier = beua_monstrous_ideas_event.4.1.m
					has_province_modifier = beua_monstrous_ideas_event.4.2.m
				}
			}
			OR = {
				has_goblin_minority_trigger = yes
				has_goblin_majority_trigger = yes
			}
		}
	}

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					NOT = {
						OR = {
							has_province_modifier = beua_monstrous_ideas_event.4.1.m
							has_province_modifier = beua_monstrous_ideas_event.4.2.m
						}
					}
					OR = {
						has_goblin_minority_trigger = yes
						has_goblin_majority_trigger = yes
					}
				}
				save_event_target_as = beua_monstrous_ideas_event_4_c
			}
		}
	}

	mean_time_to_happen = { 
		days = 1 
	}

	option = {
		name = beua_monstrous_ideas_event.4.1
		event_target:beua_monstrous_ideas_event_4_c = {
			add_province_modifier = {
				name = beua_monstrous_ideas_event.4.1.m
				duration = 3650
			}
			random_list = {
				10 = { add_base_tax = 1 }
				10 = { add_base_production = 1 }
				10 = { add_base_manpower = 1 }
			}
			if = {
				limit = { has_goblin_majority_trigger = yes }
				random = {
					chance = 50
					add_base_production = 1
				}
			}
			else_if = {
				limit = { has_goblin_minority_trigger = yes }
				add_goblin_minority_size_effect = yes
			}
		}
	}

	option = {
		name = beua_monstrous_ideas_event.4.2
		event_target:beua_monstrous_ideas_event_4_c = {
			add_province_modifier = {
				name = beua_monstrous_ideas_event.4.2.m
				duration = 3650
			}
		}
	}
}

country_event = {
	id = beua_monstrous_ideas_event.6
	title = beua_monstrous_ideas_event.6.t
	desc = beua_monstrous_ideas_event.6.d

	picture = LAND_MILITARY_eventPicture
	
	is_triggered_only = yes

	trigger = {
		has_idea_group = beua_monstrous_ideas
		is_at_war = yes
		any_owned_province = {
			NOT = {
				OR = {
					development = 20
					trade_goods = slaves
					has_terrain = dwarven_hold
					has_terrain = dwarven_hold_surface
					has_terrain = city_terrain
				}
			}
		}
	}

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					NOT = {
						OR = {
							development = 20
							trade_goods = slaves
							has_terrain = dwarven_hold
							has_terrain = dwarven_hold_surface
							has_terrain = city_terrain
						}
					}
				}
				save_event_target_as = beua_monstrous_ideas_event_6_c
			}
		}
	}

	mean_time_to_happen = { 
		days = 1
	}

	option = {
		name = beua_monstrous_ideas_event.6.1
		event_target:beua_monstrous_ideas_event_6_c = {
			change_trade_goods = slaves
		}
	}

	option = {
		name = beua_monstrous_ideas_event.6.2
		highlight = yes
		trigger = { high_tolerance_gnollish_race_trigger = yes }
		event_target:beua_monstrous_ideas_event_6_c = {
			change_trade_goods = slaves
			random_list = {
				10 = { add_base_tax = 1 }
				10 = { add_base_production = 1 }
				10 = { add_base_manpower = 1 }
			}
			if = {
				limit = { has_gnollish_majority_trigger = yes }
				random = {
					chance = 50
					add_base_production = 1
				}
			}
			else_if = {
				limit = { has_gnollish_minority_trigger = yes }
				add_gnollish_minority_size_effect = yes
			}
		}
	}

	option = {
		name = beua_monstrous_ideas_event.6.3
	}
}
