# Event 1
country_event = {
    id = beua_magocratic_ideas_event.1
	title = beua_magocratic_ideas_event.1.t
	desc = beua_magocratic_ideas_event.1.d

	picture = ADVISOR_eventPicture
	
	is_triggered_only = yes

	trigger = {
		has_idea_group = beua_magocratic_ideas
        has_estate = estate_mages
	}

	mean_time_to_happen = {
        days = 1
    }
    
	option = {
		name = beua_magocratic_ideas_event.1.1
        define_heir = {
			dynasty = ROOT
			culture = ROOT
			religion = ROOT
            claim = 25
            change_adm = 2
			change_dip = 2
			change_mil = 2
			age = 0
		}
        add_adm_power = -50
		add_dip_power = -50
		add_mil_power = -50
        set_heir_random_planetouched_effect = yes
        set_heir_mage_effect = yes
		add_stability = -1
	}

    option = {
		name = beua_magocratic_ideas_event.1.2
        add_adm_power = 25
		add_dip_power = 25
		add_mil_power = 25
		add_stability = 1
	}

	after = {
		set_country_flag = beua_magocratic_ideas_event.1.f
	}
}

# Event 5
country_event = {
	id = beua_magocratic_ideas_event.5
	title = beua_magocratic_ideas_event.5.t
	desc = beua_magocratic_ideas_event.5.d

	picture = FORT_eventPicture
	
	is_triggered_only = yes

	trigger = {
		has_idea_group = beua_magocratic_ideas
        has_estate = estate_mages
        NOT = { has_country_modifier = beua_magocratic_ideas_event.5.1.m }
        any_owned_province = {
			fort_level = 2
            NOT = { has_province_modifier = beua_magocratic_ideas_event.5.2.m }
        }
	}

	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					fort_level = 2
                    NOT = { has_province_modifier = beua_magocratic_ideas_event.5.2.m }
				}
				save_event_target_as = beua_magocratic_ideas_event.5.c
			}
		}
	}

	mean_time_to_happen = { 
        days = 1
    }

	option = {
		name = beua_magocratic_ideas_event.5.1
		add_dip_power = -25
		add_country_modifier = {
			name = beua_magocratic_ideas_event.5.1.m
			duration = 3650
		}
	}

	option = {
		name = beua_magocratic_ideas_event.5.2
		event_target:beua_magocratic_ideas_event.5.c = {
			add_province_modifier = {
				name = "beua_magocratic_ideas_event.5.2.m"
				duration = -1
			}
		}
	}
}
