beua_force_government_change = {
	category = influence

	require_acceptance = no

	is_visible = {
		has_idea = respected_authority
		FROM = {
			is_subject_of = ROOT
			is_subject_other_than_tributary_trigger = yes
			OR = {
				NOT = { dynasty = ROOT }
				NOT = { culture_group = ROOT }
				NOT = { religion = ROOT }
				NOT = { government = monarchy }
			}
		}
	}

	is_allowed = {
		if = {
			limit = { 
				FROM = { liberty_desire = 50 } 
			}
			custom_trigger_tooltip = {
				tooltip = beua_force_government_change_is_rebellious
				FROM = { NOT = { liberty_desire = 50 } }
			}
		}
		if = {
			limit = {
				has_country_modifier = beua_political_intervention_strain
			}
			custom_trigger_tooltip = {
				tooltip = beua_force_government_change_cooldown
				NOT = { has_country_modifier = beua_political_intervention_strain }
			}
		}
		FROM = {
			is_subject_of = ROOT
			is_subject_other_than_tributary_trigger = yes
			OR = {
				NOT = { dynasty = ROOT }
				NOT = { culture_group = ROOT }
				NOT = { religion = ROOT }
				NOT = { government = monarchy }
			}
		}
	}

	on_accept = {
		add_country_modifier = {
			name = beua_political_intervention_strain
			duration = 9125
		}

		if = {
			limit = {
				FROM = {
					NOT = {
						government = monarchy
					}
				}
			}
			add_mil_power = -250

			FROM = {
				add_liberty_desire = 25
				change_government = monarchy
				add_government_reform = feudalism_reform
			}
		}

		if = {
			limit = {
				FROM = {
					NOT = {
						culture_group = ROOT
					}
				}
			}
			add_dip_power = -250

			FROM = {
				add_liberty_desire = 25
				change_primary_culture = ROOT

				capital_scope = {
					change_culture = ROOT
				}
			}
		}

		if = {
			limit = {
				FROM = {
					NOT = {
						religion = ROOT
					}
				}
			}
			add_adm_power = -250

			FROM = {
				add_liberty_desire = 25
				change_religion = ROOT

				capital_scope = {
					change_religion = ROOT
				}
			}
		}

		if = {
			limit = {
				FROM = {
					NOT = {
						dynasty = ROOT
					}
				}
			}
			add_adm_power = -75
			add_dip_power = -75
			add_mil_power = -75

			FROM = {
				add_liberty_desire = 25
				define_ruler = {
					dynasty = ROOT
					culture = ROOT
					religion = ROOT
					min_age = 20
					max_age = 40
				}
				define_heir = {
					dynasty = ROOT
					culture = ROOT
					religion = ROOT
					age = 5
				}
			}
		}
	}

	ai_will_do = {
		always = no
	}
}

					OR = {
						is_subject_of_type = vassal

					}
				}
			}
		}
		if = {
			limit = { FROM = { NOT = { dynasty = ROOT } } }
			custom_trigger_tooltip = {
				tooltip = beua_appanage_create_not_same_dynasty
				FROM = { dynasty = ROOT }
			}
		}
	}
	on_accept = {
		create_subject = {
			subject_type = beua_appanage
			subject = FROM
			keep_liberty_desire = yes
		}
		FROM = {
			add_government_reform = feudalism_reform
		}
	}

	ai_will_do = {
		always = no
	}
}

# Grant Indepence to Appanage
beua_appanage_release = {
	category = influence
	require_acceptance = no
	is_visible = {
		FROM = {
			is_subject_of = ROOT
			is_subject_of_type = beua_appanage
		}
	}
	is_allowed = {
		FROM = {
			is_subject_of = ROOT
			is_subject_of_type = beua_appanage
		}
	}
	on_accept = {
		FROM = {
			if = {
				limit = { NOT = { liberty_desire = 50 } }
				add_opinion = {
					who = root
					modifier = vassal_abandoned
				}
			}
			grant_independence = yes
			every_owned_province = {
				limit = {
					or = {
						is_claim = root
						is_permanent_claim = root
						is_core = root
					}
				}
				if = {
					limit = { is_permanent_claim = root }
					remove_claim = root
				}
				if = {
					limit = { is_claim = root }
					remove_claim = root
				}
				if = {
					limit = { is_core = root }
					remove_core = root
				}
			}
		}
		add_prestige = -25
	}
	ai_will_do = {
		always = no
	}
}

# Create a Dominion
beua_dominion_create = {
	category = influence

	is_visible = {
		OR = { 
			has_idea = beua_imperialism_idea_3 
			has_idea = beua_decentralism_idea_7 
			has_idea = postal_service 
		}
		
		FROM = {
			is_subject_of = ROOT
			NOT = {
				OR = {
					is_subject_of_type = beua_dominion
					is_subject_of_type = beua_client_dominion
				}
			}
			is_subject_other_than_tributary_trigger = yes
		}
	}

	is_allowed = {
		FROM = {
			is_subject_of = ROOT
		}
		if = {
			limit = { FROM = { liberty_desire = 25 } }
			custom_trigger_tooltip = {
				tooltip = beua_dominion_create_too_much_liberty_desire
				FROM = { NOT = { liberty_desire = 25 } }
			}
		}
		if = {
			limit = {
				FROM = {
					NOT = { is_subject_of_type = vassal }
					NOT = { is_subject_of_type = client_vassal }
				}
			}
			custom_trigger_tooltip = {
				tooltip = FROM_IS_NOT_VASSAL_OF_ROOT
				FROM = {
					OR = {
						is_subject_of_type = vassal
						is_subject_of_type = client_vassal
					}
				}
			}
		}
	}

	on_accept = {
		if = { limit = { FROM = { is_subject_of_type = client_vassal } }
			ROOT = {
				create_subject = {
					subject_type = beua_client_dominion
					subject = FROM
				}
			}
			FROM = {
				set_country_flag = enables_estate_nobles_flag
				set_country_flag = enables_estate_burghers_flag
				set_country_flag = enables_estate_church_flag
				set_country_flag = has_overriden_color_flag
				set_country_flag = has_overriden_name_flag
			}
		}
		if = { limit = { FROM = { is_subject_of_type = vassal } }
			ROOT = {
				create_subject = {
					subject_type = beua_dominion
					subject = FROM
				}
			}
			FROM = {
				set_country_flag = enables_estate_nobles_flag
				set_country_flag = enables_estate_burghers_flag
				set_country_flag = enables_estate_church_flag
				set_country_flag = has_overriden_color_flag
				set_country_flag = has_overriden_name_flag
			}
		}
	}

	ai_will_do = {
		OR = {
			has_unified_culture_group = yes
			AND = {
				NOT = { has_country_modifier = human_administration }
				FROM = { has_country_modifier = human_administration } 
			}
		}
	}
}
