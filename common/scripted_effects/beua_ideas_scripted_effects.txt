# [beua_industrial_idea_7|beua_imperial_idea_6|beua_serpentspine_idea_1]
beua_apply_development_modifiers = {
    # Early Check
    if = {
        limit = {
            owner = {
                OR = {
                    has_idea = beua_industrial_idea_7
                    has_idea = beua_imperial_idea_6
                    has_idea = beua_serpentspine_idea_1
                }
            }
        }

        # [beua_industrial_idea_7]
        if = {
            limit = {
                owner = {
                    has_idea = beua_industrial_idea_7
                }
                NOT = {
                    has_province_modifier = beua_supply_chain_complex_modifier
                }
            }

            if = {
                limit = {
                    development = 30
                }

                if = {
                    limit = { 
                        has_province_modifier = beua_supply_chain_simple_modifier 
                    }
                    remove_province_modifier = beua_supply_chain_simple_modifier
                }

                add_province_modifier = {
                    name = beua_supply_chain_complex_modifier
                    duration = -1
                }

                owner = {
                    mission_reward_artifice_points_1 = yes
                }
            }
            else_if = {
                limit = {
                    development = 15
                    NOT = { has_province_modifier = beua_supply_chain_simple_modifier }
                    NOT = { has_province_modifier = beua_supply_chain_complex_modifier }
                }
                add_province_modifier = {
                    name = beua_supply_chain_simple_modifier
                    duration = -1
                }
            }
        }

        # [beua_imperial_idea_6]
        if = {
            limit = {
                owner = {
                    has_idea = beua_imperial_idea_6
                }
                NOT = {
                    has_province_modifier = beua_urban_industry
                }
            }

            if = {
                limit = {
                    development = 20
                    OR = {
                        has_terrain = dwarven_hold
                        has_terrain = dwarven_hold_surface
                        has_terrain = city_terrain
                    }
                }

                add_province_modifier = {
                    name = beua_urban_industry
                    duration = -1
                }
            }
        }

        # [beua_serpentspine_idea_1]
        if = {
            limit = {
                owner = {
                    has_idea = beua_serpentspine_idea_1
                }
                NOT = {
                    has_province_modifier = beua_subterranean_bastion
                }
            }

            if = {
                limit = {
                    fort_level = 1
                    development = 15
                    continent = serpentspine
                }

                add_province_modifier = {
                    name = beua_subterranean_bastion
                    duration = -1
                }
            }
        }
    }
}
