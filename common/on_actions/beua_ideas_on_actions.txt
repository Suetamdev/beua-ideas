# [beua_industrial_idea_7|beua_imperial_idea_6|beua_serpentspine_idea_1]
on_province_owner_change = {
    # beua_industrial_idea_7
    if = {
        limit = {
			owner = {
				has_idea = beua_industrial_idea_7
			}
			NOT = {
				OR = {
					has_province_modifier = beua_supply_chain_simple_modifier
					has_province_modifier = beua_supply_chain_complex_modifier
				}
			}
			
			development = 15
        }

        if = {
            limit = {
                development = 30
            }

            add_province_modifier = {
                name = beua_supply_chain_complex_modifier
                duration = -1
            }

            #owner = {
			#	mission_reward_artifice_points_1 = yes
			#}
        }
        else = {
            add_province_modifier = {
                name = beua_supply_chain_simple_modifier
                duration = -1
            }
        }
    } 
	else_if = {
		limit = {
			OR = {
				has_province_modifier = beua_supply_chain_complex_modifier
				has_province_modifier = beua_supply_chain_simple_modifier
			}
			owner = {
				NOT = {
					has_idea = beua_industrial_idea_7
				}  
			}
		}
		beua_clear_supply_chain_modifiers = yes
	}

	# beua_imperial_idea_6
	if = {
		limit = {
			owner = { 
				has_idea = beua_imperial_idea_6 
			}
			NOT = {
				has_province_modifier = beua_urban_industry
			}
			development = 20
			OR = {
				has_terrain = dwarven_hold
				has_terrain = dwarven_hold_surface
				has_terrain = city_terrain
			}
		}
		hidden_effect = {
			add_province_modifier = { 
				name = beua_urban_industry
				duration = -1
			}
		}
	}
	else_if = {
		limit = {
			has_province_modifier = beua_urban_industry
			owner = { 
				NOT = { has_idea = beua_imperial_idea_6 }
			}
		}
		hidden_effect = {
			remove_province_modifier = beua_urban_industry
		}
	}

	# beua_serpentspine_idea_1
	if = {
		limit = {
			owner = { 
				has_idea = beua_serpentspine_idea_1 
			}
			NOT = {
				has_province_modifier = beua_subterranean_bastion
			}
			fort_level = 1
			development = 15
			continent = serpentspine
		}
		hidden_effect = {
			add_province_modifier = { 
				name = beua_subterranean_bastion
				duration = -1
			}
		}
	}
	else_if = {
		limit = {
			has_province_modifier = beua_subterranean_bastion
			owner = { 
				NOT = { has_idea = beua_serpentspine_idea_1 }
			}
		}
		hidden_effect = {
			remove_province_modifier = beua_subterranean_bastion
		}
	}
}

# [beua_legionary_idea_4]
on_province_religion_changed = {
	if = {
		limit = {
			has_owner_religion = yes
			owner = {
				has_idea = beua_legionary_idea_4
			}
		}
		export_to_variable = { which = beua_province_development value = trigger_value:development who = ROOT }
		while = { limit = { check_variable = { which = beua_province_development value = 1 } }
			owner = { add_manpower = 0.3 }
			change_variable = { which = beua_province_development value = -1 }
		}
	}
}
