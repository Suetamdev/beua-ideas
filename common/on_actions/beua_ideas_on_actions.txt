# [state_administration_idea_5]
on_centralized_state = {
	if = {
        limit = {
            FROM = {
				has_idea = state_administration_idea_5
			}
        }
		area = {
			limit = {
				owned_by = FROM
			}
			random_list = {
				1 = { add_base_tax = 1 }
				1 = { add_base_production = 1 }
				1 = { add_base_manpower = 1 }
			}
		}
	}
}

# [industrial_idea_7|imperial_idea_6]
on_adm_development = {
    # Early Check
    if = {
        limit = {
            FROM = {
                OR = {
                    has_idea = industrial_idea_7
                    has_idea = imperial_idea_6
                }
            }
        }
    }

    # [industrial_idea_7]
    if = {
        limit = {
            FROM = {
				has_idea = industrial_idea_7
			}
            NOT = {
                has_province_modifier = supply_chain_complex_modifier
            }
        }

        if = {
            limit = {
                development = 50
            }

            if = {
                limit = { 
                    has_province_modifier = supply_chain_simple_modifier 
                }
                remove_province_modifier = supply_chain_simple_modifier
            }

            add_province_modifier = {
                name = supply_chain_complex_modifier
                duration = -1
            }

            owner = {
				mission_reward_artifice_points_1 = yes
			}
        }
        else_if = {
            limit = {
                development = 25
                NOT = { has_province_modifier = supply_chain_simple_modifier }
                NOT = { has_province_modifier = supply_chain_complex_modifier }
            }
            add_province_modifier = {
                name = supply_chain_simple_modifier
                duration = -1
            }
        }
    }

    # [imperial_idea_6]
    if = {
        limit = {
            FROM = {
				has_idea = imperial_idea_6
			}
            NOT = {
                has_province_modifier = urban_industry
            }
        }

        if = {
            limit = {
                development = 25
                OR = {
					has_terrain = dwarven_hold
					has_terrain = dwarven_hold_surface
					has_terrain = city_terrain
				}
            }

            add_province_modifier = {
                name = urban_industry
                duration = -1
            }

        }
    }
}

# [industrial_idea_7|imperial_idea_6]
on_province_owner_change = {
	# imperial_idea_6
	if = {
		limit = {
			NOT = {
				has_province_modifier = urban_industry
			}
			AND = {
				owner = { has_idea = imperial_idea_6 }
			}
		}
		hidden_effect = {
			owner = { set_country_flag = has_urban_industry }
			if = {
				limit = {
					OR = {
						has_terrain = dwarven_hold
						has_terrain = dwarven_hold_surface
						has_terrain = city_terrain
					}
				}
				add_province_modifier = { 
					name = "urban_industry"
					duration = -1
				}
			}
		}
	}

    # industrial_idea_7
    if = {
        limit = {
            FROM = {
				has_idea = industrial_idea_7
			}
            NOT = {
                OR = {
                    has_province_modifier = supply_chain_complex_modifier
                    has_province_modifier = supply_chain_simple_modifier
                }
            }
        }

        if = {
            limit = {
                development = 50
            }

            add_province_modifier = {
                name = supply_chain_complex_modifier
                duration = -1
            }

            owner = {
				mission_reward_artifice_points_1 = yes
			}
        }
        else_if = {
            limit = {
                development = 25
            }
            add_province_modifier = {
                name = supply_chain_simple_modifier
                duration = -1
            }
        }
    }
}
